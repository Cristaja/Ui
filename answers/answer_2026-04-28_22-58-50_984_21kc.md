Ошибка связана с тем, что GitHub отклонил коммит, потому что в тексте ответа ассистента оказалась строка, которая выглядит как токен (секрет). GitHub автоматически сканирует содержимое и блокирует пуши при обнаружении подозрительных строк. Однако файл всё равно сохранился — скрипт повторил попытку, когда текст ответа изменился (или вы задали новый вопрос). Это говорит о том, что в целом скрипт работает, но иногда ответы могут содержать фрагменты, вызывающие ложное срабатывание сканера секретов.Я доработал скрипт, чтобы он сам очищал текст от потенциальных секретов (удаляет строки, похожие на токены, ключи, API‑ключи) перед отправкой на GitHub. Это предотвратит подобные ошибки в будущем.Обновлённый скрипт (с фильтрацией секретов)javascriptCopyDownload// ==UserScript==
// @name         DeepSeek → Cristaja/Ui (Clean)
// @namespace    http://tampermonkey.net/
// @version      5.5
// @description  Сохраняет ответы в репозиторий, очищая от потенциальных секретов
// @match        https://chat.deepseek.com/*
// @grant        GM_xmlhttpRequest
// @connect      api.github.com
// ==/UserScript==

(function() {
    'use strict';

    const GITHUB_TOKEN = '[POTENTIAL_SECRET_REMOVED]';
    const REPO_OWNER   = 'Cristaja';
    const REPO_NAME    = 'Ui';
    const BRANCH       = 'main';
    const BASE_FOLDER  = 'answers/';

    const processed = new WeakSet();

    // Удаляем строки, которые могут быть приняты за токены/ключи
    function sanitizeText(text) {
        // Удаляем всё, что похоже на GitHub токен классического типа (ghp_... или github_pat_...)
        text = text.replace(/ghp_[a-zA-Z0-9]{36,}/g, '[GITHUB_TOKEN_REMOVED]');
        // Удаляем общие паттерны API-ключей (например, sk-..., key-...)
        text = text.replace(/\b[a-zA-Z0-9_-]{30,}\b/g, (match) => {
            // Если очень высокая энтропия (много разных символов), заменяем
            if (/^(ghp_|github_pat_|sk-|key-|api-|sec-|token-)/i.test(match) ||
                (match.length > 35 && /[A-Z]/.test(match) && /[a-z]/.test(match) && /\d/.test(match))) {
                return '[POTENTIAL_SECRET_REMOVED]';
            }
            return match;
        });
        return text;
    }

    function getFinalAnswerText(container) {
        const blocks = container.querySelectorAll('.ds-markdown');
        if (blocks.length === 0) return null;
        const lastBlock = blocks[blocks.length - 1];
        const clone = lastBlock.cloneNode(true);
        clone.querySelectorAll('details').forEach(d => d.remove());
        const rawText = clone.textContent?.trim();
        return rawText ? sanitizeText(rawText) : null;
    }

    function generateFileName() {
        const now = new Date();
        const pad = (n) => String(n).padStart(2, '0');
        const ts = `${now.getFullYear()}-${pad(now.getMonth()+1)}-${pad(now.getDate())}_${pad(now.getHours())}-${pad(now.getMinutes())}-${pad(now.getSeconds())}`;
        return `answer_${ts}.md`;
    }

    function commitAnswerFile(text) {
        const filePath = BASE_FOLDER + generateFileName();
        const contentBase64 = btoa(unescape(encodeURIComponent(text)));
        const url = `https://api.github.com/repos/${REPO_OWNER}/${REPO_NAME}/contents/${filePath}`;

        GM_xmlhttpRequest({
            method: 'PUT',
            url: url,
            headers: {
                'Authorization': `token ${GITHUB_TOKEN}`,
                'Accept': 'application/vnd.github.v3+json',
                'Content-Type': 'application/json'
            },
            data: JSON.stringify({
                message: `Add DeepSeek answer [${filePath}]`,
                content: contentBase64,
                branch: BRANCH
            }),
            onload: function(response) {
                if (response.status === 201 || response.status === 200) {
                    const data = JSON.parse(response.responseText);
                    console.log('📁 Файл сохранён:', data.content.html_url);
                } else {
                    const err = JSON.parse(response.responseText);
                    if (err.message?.includes('Secret detected')) {
                        console.warn('⚠️ GitHub заблокировал коммит из-за подозрения на секрет. Файл не сохранён. Попробуем позже.');
                    } else {
                        console.error('❌ Ошибка сохранения:', response.responseText);
                    }
                }
            },
            onerror: (err) => console.error('❌ Сеть:', err)
        });
    }

    function processAnswer(container) {
        const text = getFinalAnswerText(container);
        if (!text) return false;
        commitAnswerFile(text);
        return true;
    }

    function handleContainer(container) {
        if (processed.has(container)) return;
        processed.add(container);
        if (processAnswer(container)) return;
        const obs = new MutationObserver(() => {
            if (processAnswer(container)) obs.disconnect();
        });
        obs.observe(container, { childList: true, subtree: true });
        setTimeout(() => obs.disconnect(), 30000);
    }

    function handleNode(node) {
        const msg = node.closest?.('.ds-message');
        if (msg) handleContainer(msg);
    }

    const mainObs = new MutationObserver(muts => {
        for (const m of muts) {
            for (const node of m.addedNodes) {
                if (node.nodeType !== Node.ELEMENT_NODE) continue;
                if (node.matches?.('.ds-message')) handleContainer(node);
                if (node.querySelectorAll) {
                    if (node.querySelectorAll('.ds-markdown, button').length > 0) handleNode(node);
                    node.querySelectorAll('.ds-message').forEach(handleContainer);
                }
            }
        }
    });

    mainObs.observe(document.body, { childList: true, subtree: true });
    document.querySelectorAll('.ds-message').forEach(handleContainer);
    console.log('🤖 Авто-сохранение (с очисткой секретов) запущено.');
})();Что изменилосьДобавлена функция sanitizeText, которая удаляет строки, похожие на токены и ключи, чтобы не нарушать политику GitHub.Улучшена обработка ошибки Secret detected: теперь она не выводится как красная ошибка, а предупреждение.Файл сохраняется только после очистки, что практически исключает блокировку.ПроверкаУстановите этот обновлённый скрипт (замените предыдущий).Задайте вопрос, который может содержать что-то похожее на секрет (например, случайно сгенерированный ключ). Либо просто продолжайте пользоваться — теперь ответы будут автоматически сохраняться без ошибок.Убедитесь, что в папке answers/ появляются новые .md файлы.Если ошибка с секретами возникала только однажды, очистка текста решит проблему на будущее. Но это снизит безопасность. Лучше использовать очистку текста, как в обновлённом скрипте.
